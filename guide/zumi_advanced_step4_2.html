/* HTML — videoCanvas 위에 투명 aiCanvas 겹치기 */ <div style="position:relative"> <canvas id="videoCanvas"></canvas> <canvas id="aiCanvas" style="position:absolute;inset:0;pointer-events:none"></canvas> </div> // JS — AI 루프 (120ms마다 canvas 캡처 → AI 전달) async function aiLoop() { aiCtx.clearRect(0, 0, aiCanvas.width, aiCanvas.height); const snap = document.createElement('canvas'); snap.getContext('2d').drawImage(videoCanvas, 0, 0); if (faceOn) await faceMesh.send({image: snap}); if (handsOn) await hands.send({image: snap}); if (cocoOn) await runCoco(snap); setTimeout(aiLoop, 150); }

const faceMesh = new FaceMesh({ locateFile: f => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${f}` }); faceMesh.setOptions({ maxNumFaces: 2, refineLandmarks: false, minDetectionConfidence: 0.5 }); faceMesh.onResults((results) => { if (!results.multiFaceLandmarks?.length) return; for (const lm of results.multiFaceLandmarks) { drawConnectors(aiCtx, lm, FACEMESH_FACE_OVAL, {color: 'rgba(139,92,246,.6)', lineWidth: 2}); drawConnectors(aiCtx, lm, FACEMESH_LEFT_EYE, {color: 'rgba(6,182,212,.8)', lineWidth: 1.5}); } // 얼굴 중심 X 좌표 (0~1) → 로봇 제어에 활용 const faceX = results.multiFaceLandmarks[0][1].x; // 0=왼쪽, 1=오른쪽 if (faceX < 0.4) sendRaw(buildPacket(52, 3)); // 좌회전 else if (faceX > 0.6) sendRaw(buildPacket(53, 3)); // 우회전 });
✋ MediaPipe Hands — 손 제스처
const hands = new Hands({ locateFile: f => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${f}` }); hands.setOptions({ maxNumHands: 1, modelComplexity: 0, minDetectionConfidence: 0.6 }); hands.onResults((results) => { if (!results.multiHandLandmarks?.length) return; const lm = results.multiHandLandmarks[0]; drawConnectors(aiCtx, lm, HAND_CONNECTIONS, {color: 'rgba(6,182,212,.9)', lineWidth: 2}); // 손가락 펴짐 여부로 제스처 판별 const fingers = getFingers(lm); // [엄지,검지,중지,약지,소지] const count = fingers.filter(Boolean).length; if (count === 5) sendRaw(buildPacket(29, 1, 0, 0)); // 손바닥 → 전진 if (count === 0) sendRaw(buildPacket(25)); // 주먹 → 정지 });
🚦 COCO-SSD — 물체·표지판 감지
// 모델 로드 (처음 1회) let cocoModel; cocoSsd.load({ base: 'lite_mobilenet_v2' }).then(m => cocoModel = m); // 감지 및 박스 그리기 async function runCoco(srcCanvas) { const preds = await cocoModel.detect(srcCanvas); for (const p of preds.filter(p => p.score > 0.45)) { const [x, y, w, h] = p.bbox; aiCtx.strokeStyle = p.class === 'stop sign' ? '#ef4444' : '#8b5cf6'; aiCtx.lineWidth = 2; aiCtx.strokeRect(x, y, w, h); aiCtx.fillStyle = aiCtx.strokeStyle; aiCtx.fillText(`${p.class} ${Math.round(p.score*100)}%`, x, y - 5); // 정지 표지판 감지 시 로봇 정지 if (p.class === 'stop sign') sendRaw(buildPacket(25)); } }
🎓 Teachable Machine — 나만의 분류기
// Teachable Machine 모델 URL 입력 후 로드 const MODEL_URL = 'https://teachablemachine.withgoogle.com/models/YOUR_MODEL_ID/'; let tmModel; tmImage.load(MODEL_URL + 'model.json', MODEL_URL + 'metadata.json') .then(m => tmModel = m); // 분류 실행 async function runTeachable(srcCanvas) { const preds = await tmModel.predict(srcCanvas); const top = preds.sort((a,b) => b.probability - a.probability)[0]; if (top.probability > 0.8) { // 클래스 이름에 따라 로봇 명령 전송 if (top.className === 'forward') sendRaw(buildPacket(29, 1, 0, 0)); if (top.className === 'stop') sendRaw(buildPacket(25)); } }

# ZUMI 로봇 AI 비전 웹앱 (브라우저 AI 방식) ## 시스템 구조 브라우저(HTML 단독) ↔ WebSocket(ws://IP/ws) ↔ ESP32(주미) 서버 설치 없음. HTML 파일 하나로만 동작. AI 처리도 브라우저에서 직접 수행 (CDN 라이브러리 사용). ## 카메라 스트리밍 (기존과 동일) ws.binaryType = 'arraybuffer' ws.send('stream') // 연결 후 전송 수신: JPEG 바이너리 → Blob → Image → canvas.drawImage ## AI 오버레이 구조 videoCanvas (영상) + aiCanvas (AI 결과, position:absolute, pointer-events:none) 120~200ms 간격으로 videoCanvas 캡처 → AI 모델에 전달 → aiCanvas에 결과 그리기 ## CDN 라이브러리 (jsdelivr.net) // MediaPipe 얼굴 @mediapipe/face_mesh, @mediapipe/drawing_utils // MediaPipe 손 @mediapipe/hands // TF.js + COCO-SSD (물체/표지판) @tensorflow/tfjs@4.17.0, @tensorflow-models/coco-ssd@2.2.3 // Teachable Machine @tensorflow/tfjs@4.17.0, @teachablemachine/image@0.8.5 ## 이동 명령 (패킷 구조) function buildPacket(cmd, ...params) { const buf = new Uint8Array(4 + params.length); buf[0]=0x24; buf[1]=0x52; buf[2]=cmd; buf[3]=0x00; params.forEach((v,i)=>buf[4+i]=v); return buf.buffer; } 전진(무한): buildPacket(29, speed, 0, 0) 정지: buildPacket(25) 좌회전: buildPacket(52, deg/5) 우회전: buildPacket(53, deg/5) LED: buildPacket(10, r, g, b) // r,g,b 각 0~10 ## 기본 UI 포함 요소 - IP 입력 + 연결 버튼 + 상태 표시 - 영상 스트리밍 canvas (640×480) ws.binaryType='arraybuffer', 연결 후 ws.send('stream') 수신: Blob → Image → canvas.drawImage ## 결과물 조건 - HTML 파일 하나로 동작 - CSS/JavaScript 내부 포함 - Chrome에서 바로 실행 가능 - 별도 서버 불필요 ## 추가할 AI 기능: (여기에 원하는 것을 적어주세요) 예시: - MediaPipe FaceMesh 얼굴 감지 토글 버튼 + 랜드마크 오버레이 - 손 제스처로 전진/정지/좌우회전 제어 - COCO-SSD 물체 인식 + stop sign 감지 시 자동 정지 - 얼굴이 왼쪽이면 좌회전, 오른쪽이면 우회전 (얼굴 추적)
